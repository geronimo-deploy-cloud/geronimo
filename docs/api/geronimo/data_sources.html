geronimo.data_sources

Geronimo Data Layer.

The data_sources module provides a unified abstraction for connecting to, querying, and ingesting data from various upstream sources. It decouples the ML modeling logic from the underlying data infrastructure.

Key components:

  • DataSource: Represents a table or view in a database.
  • Query: A composable query object for retrieving data.
  • DatabaseConnection: Protocol for database adapters.

Supported connections:

  • Snowflake
  • PostgreSQL
  • SQL Server
  • Google BigQuery (via generic interface)

This layer handles connection pooling, query generation, and data type mapping.

107        self.source = SourceType(source) if isinstance(source, str) else source
108        self.query = query
109        self.path = path
110        self.handle = handle
111        self.connection_params = connection_params or {}
112        self._custom_connection = connection

116        if self.source == SourceType.FUNC:

129    def load(self, **params) -> pd.DataFrame:
130        """Load data from source.
131
132        Args:
133            **params: Parameters passed to the data loading function.
134                      For database sources, these are query parameters.
135                      For function sources, these are passed to the handle.
136
137        Returns:
138            DataFrame with loaded data.
139        
140        Raises:
141            DataSourceError: If function source doesn't return a DataFrame.
142        """
143        if self.source == SourceType.FILE:
144            return self._load_file()
145        elif self.source == SourceType.FUNC:
146            return self._load_function(**params)
147        else:
148            return self._load_database(**params)

150    def _load_function(self, **params) -> pd.DataFrame:
151        """Load data by calling the handle function.
152        
153        Validates at runtime that the function returns a DataFrame.
154        
155        Args:
156            **params: Keyword arguments passed to the handle function.
157        
158        Returns:
159            DataFrame returned by the handle function.
160        
161        Raises:
162            DataSourceError: If handle doesn't return a DataFrame or raises an exception.
163        """
164        try:
165            result = self.handle(**params)
166        except Exception as e:
167            raise DataSourceError(
168                f"DataSource '{self.name}' handle function raised an exception: {e}"
169            ) from e
170        
171        # Runtime validation: ensure result is a DataFrame
172        if not isinstance(result, pd.DataFrame):
173            actual_type = type(result).__name__
174            raise DataSourceError(
175                f"DataSource '{self.name}' handle function must return a pandas DataFrame, "
176                f"but returned {actual_type}. "
177                f"Ensure your function returns pd.DataFrame."
178            )
179        
180        return result
181
182    def _load_file(self) -> pd.DataFrame:
183        """Load data from file."""
184        from pathlib import Path
185
186        path = Path(self.path)
187        if path.suffix == ".csv":
188            return pd.read_csv(path)
189        elif path.suffix in [".parquet", ".pq"]:
190            return pd.read_parquet(path)
191        elif path.suffix == ".json":
192            return pd.read_json(path)
193        else:
194            raise ValueError(f"Unsupported file format: {path.suffix}")
195
196    def _load_database(self, **params) -> pd.DataFrame:
197        """Load data from database using connection interface."""
198        sql = self.query.render(**params)
199        
200        # Use custom connection if provided, otherwise create from factory
201        if self._custom_connection is not None:
202            connection = self._custom_connection
203        else:
204            connection = get_connection(self.source.value, self.connection_params)
205        
206        # Use context manager for automatic connection cleanup
207        with connection:
208            return connection.execute(sql)
209
210    def __repr__(self) -> str:
211        return f"DataSource({self.name}, source={self.source.value})"

257    """Collect all DataSource objects whose variable names start with prefix.

275    
276    Args:
277        module: The module to search (typically sys.modules[__name__]).

279    
280    Returns:
281        List of DataSource objects whose variable names start with prefix.
282    """
283    sources = []
284    for name in dir(module):

1941def _no_init_or_replace_init(self, *args, **kwargs):
1942    cls = type(self)
1943
1944    if cls._is_protocol:
1945        raise TypeError('Protocols cannot be instantiated')
1946
1947    # Already using a custom `__init__`. No need to calculate correct
1948    # `__init__` to call. This can lead to RecursionError. See bpo-45121.
1949    if cls.__init__ is not _no_init_or_replace_init:
1950        return
1951
1952    # Initially, `__init__` of a protocol subclass is set to `_no_init_or_replace_init`.
1953    # The first instantiation of the subclass will call `_no_init_or_replace_init` which
1954    # searches for a proper new `__init__` in the MRO. The new `__init__`
1955    # replaces the subclass' old `__init__` (ie `_no_init_or_replace_init`). Subsequent
1956    # instantiation of the protocol subclass will thus use the new
1957    # `__init__` and no longer call `_no_init_or_replace_init`.
1958    for base in cls.__mro__:
1959        init = base.__dict__.get('__init__', _no_init_or_replace_init)
1960        if init is not _no_init_or_replace_init:
1961            cls.__init__ = init
1962            break
1963    else:
1964        # should not happen
1965        cls.__init__ = object.__init__
1966
1967    cls.__init__(self, *args, **kwargs)
