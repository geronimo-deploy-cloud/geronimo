class DeterministicAnalyzer(geronimo.analyzers.Analyzer): 
 24class DeterministicAnalyzer(Analyzer):
 25    """Rule-based project analyzer using AST parsing.
 26
 27    Analyzes projects by:
 28    1. Detecting ML framework from dependencies
 29    2. Parsing Python files to find FastAPI endpoints
 30    3. Tracing service imports to build preprocessing chain
 31    4. Identifying model loading patterns
 32    """
 33
 34    # Dependency patterns for framework detection
 35    FRAMEWORK_PATTERNS: dict[MLFramework, list[str]] = {
 36        MLFramework.SKLEARN: ["scikit-learn", "sklearn"],
 37        MLFramework.PYTORCH: ["torch", "pytorch"],
 38        MLFramework.TENSORFLOW: ["tensorflow", "keras"],
 39        MLFramework.XGBOOST: ["xgboost"],
 40    }
 41
 42    # FastAPI decorator patterns
 43    ROUTE_DECORATORS = {"get", "post", "put", "delete", "patch", "options", "head"}
 44
 45    @property
 46    def analyzer_type(self) -> str:
 47        return "deterministic"
 48
 49    def analyze(self, project: ProjectScan) -> AnalysisResult:
 50        """Analyze the project using deterministic rules."""
 51        # Detect framework from dependencies
 52        framework = self._detect_framework(project.dependencies)
 53
 54        # Analyze endpoints
 55        endpoints = self._find_endpoints(project)
 56
 57        # Analyze services
 58        services = self._analyze_services(project)
 59
 60        # Build preprocessing chain
 61        preprocessing = self._build_preprocessing_chain(services, endpoints)
 62
 63        # Analyze model artifacts
 64        artifacts = self._analyze_model_artifacts(project, framework)
 65
 66        # Calculate confidence and identify items needing review
 67        confidence, needs_review, warnings = self._assess_confidence(
 68            framework, endpoints, services, artifacts
 69        )
 70
 71        return AnalysisResult(
 72            detected_framework=framework,
 73            python_version=project.python_version,
 74            dependencies=project.dependencies,
 75            endpoints=endpoints,
 76            services=services,
 77            preprocessing_chain=preprocessing,
 78            model_artifacts=artifacts,
 79            confidence=confidence,
 80            needs_review=needs_review,
 81            warnings=warnings,
 82            analyzer_type=self.analyzer_type,
 83        )
 84
 85    def _detect_framework(self, dependencies: list[str]) -> MLFramework:
 86        """Detect ML framework from dependencies."""
 87        dep_lower = [d.lower() for d in dependencies]
 88        dep_str = " ".join(dep_lower)
 89
 90        for framework, patterns in self.FRAMEWORK_PATTERNS.items():
 91            for pattern in patterns:
 92                if pattern in dep_str:
 93                    return framework
 94
 95        return MLFramework.CUSTOM
 96
 97    def _find_endpoints(self, project: ProjectScan) -> list[EndpointInfo]:
 98        """Find FastAPI endpoints by parsing route files."""
 99        endpoints = []
100
101        # Look in routes directory first, then api directory
102        search_dirs = []
103        if project.routes_dir:
104            search_dirs.append(project.routes_dir)
105        if project.api_dir:
106            search_dirs.append(project.api_dir)
107        if project.src_package:
108            search_dirs.append(project.src_package)
109
110        for search_dir in search_dirs:
111            if not search_dir or not search_dir.exists():
112                continue
113
114            for py_file in search_dir.rglob("*.py"):
115                try:
116                    file_endpoints = self._parse_endpoints_from_file(py_file)
117                    endpoints.extend(file_endpoints)
118                except Exception:
119                    # Skip files that can't be parsed
120                    continue
121
122        return endpoints
123
124    def _parse_endpoints_from_file(self, file_path: Path) -> list[EndpointInfo]:
125        """Parse a Python file to find FastAPI route decorators."""
126        endpoints = []
127
128        try:
129            source = file_path.read_text()
130            tree = ast.parse(source)
131        except (SyntaxError, UnicodeDecodeError):
132            return []
133
134        # Find the router variable name (e.g., router = APIRouter())
135        router_names = {"app", "router"}
136
137        for node in ast.walk(tree):
138            if isinstance(node, ast.Assign):
139                for target in node.targets:
140                    if isinstance(target, ast.Name):
141                        if isinstance(node.value, ast.Call):
142                            call_name = self._get_call_name(node.value)
143                            if call_name in ("APIRouter", "FastAPI"):
144                                router_names.add(target.id)
145
146        # Find decorated functions
147        for node in ast.walk(tree):
148            if isinstance(node, ast.FunctionDef) or isinstance(node, ast.AsyncFunctionDef):
149                for decorator in node.decorator_list:
150                    endpoint_info = self._parse_route_decorator(
151                        decorator, node, file_path, router_names
152                    )
153                    if endpoint_info:
154                        endpoints.append(endpoint_info)
155
156        return endpoints
157
158    def _parse_route_decorator(
159        self,
160        decorator: ast.expr,
161        func: ast.FunctionDef | ast.AsyncFunctionDef,
162        file_path: Path,
163        router_names: set[str],
164    ) -> EndpointInfo | None:
165        """Parse a route decorator to extract endpoint info."""
166        # Handle @router.get("/path") style
167        if isinstance(decorator, ast.Call):
168            if isinstance(decorator.func, ast.Attribute):
169                if (
170                    isinstance(decorator.func.value, ast.Name)
171                    and decorator.func.value.id in router_names
172                    and decorator.func.attr in self.ROUTE_DECORATORS
173                ):
174                    # Extract path from first argument
175                    path = "/"
176                    if decorator.args:
177                        if isinstance(decorator.args[0], ast.Constant):
178                            path = decorator.args[0].value
179
180                    # Extract schemas from decorator kwargs
181                    request_schema = None
182                    response_schema = None
183                    for kw in decorator.keywords:
184                        if kw.arg == "response_model":
185                            response_schema = self._get_annotation_name(kw.value)
186
187                    # Try to get request schema from function parameters
188                    for arg in func.args.args:
189                        if arg.annotation and arg.arg not in ("self", "request"):
190                            request_schema = self._get_annotation_name(arg.annotation)
191                            break
192
193                    return EndpointInfo(
194                        path=path,
195                        method=decorator.func.attr.upper(),
196                        handler_function=func.name,
197                        file_path=file_path,
198                        request_schema=request_schema,
199                        response_schema=response_schema,
200                        description=ast.get_docstring(func),
201                    )
202
203        return None
204
205    def _get_call_name(self, node: ast.Call) -> str:
206        """Get the name of a function being called."""
207        if isinstance(node.func, ast.Name):
208            return node.func.id
209        if isinstance(node.func, ast.Attribute):
210            return node.func.attr
211        return ""
212
213    def _get_annotation_name(self, node: ast.expr) -> str | None:
214        """Get the name from a type annotation."""
215        if isinstance(node, ast.Name):
216            return node.id
217        if isinstance(node, ast.Attribute):
218            return node.attr
219        if isinstance(node, ast.Subscript):
220            return self._get_annotation_name(node.value)
221        return None

223    def _analyze_services(self, project: ProjectScan) -> list[ServiceInfo]:
224        """Analyze service modules."""
225        services = []
226
227        if not project.services_dir or not project.services_dir.exists():
228            return services
229
230        for py_file in project.services_dir.glob("*.py"):
231            if py_file.name.startswith("_"):
232                continue
233
234            try:
235                service = self._parse_service_file(py_file, project)
236                if service.functions or service.classes:
237                    services.append(service)
238            except Exception:
239                continue
240
241        return services
242
243    def _parse_service_file(self, file_path: Path, project: ProjectScan) -> ServiceInfo:
244        """Parse a service file to extract functions and classes."""
245        source = file_path.read_text()
246        tree = ast.parse(source)
247
248        functions = []
249        classes = []
250        dependencies = []
251
252        for node in ast.walk(tree):
253            # Find public functions
254            if isinstance(node, ast.FunctionDef) or isinstance(node, ast.AsyncFunctionDef):
255                if not node.name.startswith("_"):
256                    functions.append(node.name)
257
258            # Find classes
259            if isinstance(node, ast.ClassDef):
260                if not node.name.startswith("_"):
261                    classes.append(node.name)
262
263            # Find imports from project
264            if isinstance(node, ast.ImportFrom):
265                if node.module and project.project_name.replace("-", "_") in node.module:
266                    dependencies.append(node.module)
267
268        return ServiceInfo(
269            name=file_path.stem,
270            file_path=file_path,
271            functions=functions,
272            classes=classes,
273            dependencies=dependencies,
274        )

276    def _build_preprocessing_chain(
277        self,
278        services: list[ServiceInfo],
279        endpoints: list[EndpointInfo],
280    ) -> list[PreprocessingStep]:
281        """Build preprocessing chain from services.
282
283        Uses heuristics to determine the order of preprocessing steps.
284        """
285        steps = []
286        order = 0
287
288        # Common preprocessing-related terms (in likely order)
289        preprocessing_patterns = [
290            ("load", "data loading"),
291            ("fetch", "data fetching"),
292            ("clean", "data cleaning"),
293            ("preprocess", "preprocessing"),
294            ("transform", "transformation"),
295            ("feature", "feature engineering"),
296            ("encode", "encoding"),
297            ("scale", "scaling"),
298            ("normalize", "normalization"),
299            ("predict", "prediction"),
300            ("infer", "inference"),
301        ]
302
303        for service in services:
304            # Check if this service is preprocessing-related
305            service_name_lower = service.name.lower()
306
307            for pattern, description in preprocessing_patterns:
308                if pattern in service_name_lower:
309                    # Add functions that match the pattern
310                    for func in service.functions:
311                        steps.append(
312                            PreprocessingStep(
313                                name=f"{service.name}.{func}",
314                                source_file=service.file_path,
315                                source_function=func,
316                                order=order,
317                            )
318                        )
319                    order += 1
320                    break
321
322        # Sort by order
323        steps.sort(key=lambda s: s.order)
324        return steps

326    def _analyze_model_artifacts(
327        self,
328        project: ProjectScan,
329        framework: MLFramework,
330    ) -> list[ModelArtifactInfo]:
331        """Analyze model artifact files."""
332        artifacts = []
333
334        for path in project.model_artifacts:
335            try:
336                size = path.stat().st_size
337            except OSError:
338                size = 0
339
340            # Detect framework from extension
341            artifact_framework = framework
342            ext = path.suffix.lower()
343            if ext in (".pt", ".pth"):
344                artifact_framework = MLFramework.PYTORCH
345            elif ext in (".h5", ".keras"):
346                artifact_framework = MLFramework.TENSORFLOW
347
348            # Determine if this is the primary model
349            is_primary = len(project.model_artifacts) == 1 or "model" in path.stem.lower()
350
351            artifacts.append(
352                ModelArtifactInfo(
353                    path=path,
354                    framework=artifact_framework,
355                    size_bytes=size,
356                    is_primary=is_primary,
357                )
358            )
359
360        return artifacts

362    def _assess_confidence(
363        self,
364        framework: MLFramework,
365        endpoints: list[EndpointInfo],
366        services: list[ServiceInfo],
367        artifacts: list[ModelArtifactInfo],
368    ) -> tuple[float, list[str], list[str]]:
369        """Assess confidence in the analysis and identify items needing review."""
370        confidence = 1.0
371        needs_review = []
372        warnings = []
373
374        # Framework detection
375        if framework == MLFramework.CUSTOM:
376            confidence -= 0.2
377            needs_review.append("Could not detect ML framework - please specify manually")
378
379        # Endpoint detection
380        if not endpoints:
381            confidence -= 0.1
382            warnings.append("No API endpoints detected")
383
384        # Check for prediction endpoint
385        has_predict = any("predict" in e.path.lower() or "predict" in e.handler_function.lower() for e in endpoints)
386        if endpoints and not has_predict:
387            warnings.append("No prediction endpoint found - is this an ML serving API?")
388
389        # Services
390        if not services:
391            confidence -= 0.1
392            warnings.append("No services directory found")
393
394        # Model artifacts
395        if not artifacts:
396            confidence -= 0.2
