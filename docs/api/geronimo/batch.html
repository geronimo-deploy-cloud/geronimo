 13    """Base class for batch ML pipelines.
 14
 15    Provides a standardized interface for batch jobs with
 16    schedule/trigger support and integrated artifact management.
 17
 18    Example:
 19        ```python
 20        from geronimo.batch import BatchPipeline, Schedule
 21        from myproject.models import CreditRiskModel
 22
 23        class DailyScoringPipeline(BatchPipeline):
 24            model_class = CreditRiskModel
 25            schedule = Schedule.daily(hour=6)
 26
 27            def run(self):
 28                # Load data
 29                data = self.model.features.data_source.load()
 30
 31                # Transform features
 32                X = self.model.features.transform(data)
 33
 34                # Predict
 35                predictions = self.model.predict(X)
 36
 37                # Save results
 38                self.save_results(predictions)
 39
 40        # Execute
 41        pipeline = DailyScoringPipeline()
 42        pipeline.initialize()
 43        pipeline.execute()
 44        ```
 45    """

113            results: Results to save (DataFrame, dict, etc.).
114            output_path: Optional output path (for file-based storage).
115            use_artifact_store: If True, save via ArtifactStore for consistency

119                          Defaults to pipeline class name.
120
121        Returns:
122            Path or URI where results were saved.
123        """
124        import pandas as pd
125        from datetime import datetime
126        from pathlib import Path
127
128        # Use ArtifactStore if requested
129        if use_artifact_store:
130            from geronimo.artifacts import ArtifactStore
131            
132            project = self.artifact_project or (
133                self.model_class.name if self.model_class else self.__class__.__name__
134            )
135            version = self.artifact_version or (
136                self.model_class.version if self.model_class else "1.0.0"
137            )
138            
139            store = ArtifactStore(project=project, version=version)
140            name = artifact_name or f"{self.__class__.__name__}_results"
141            return store.save(name, results, artifact_type="pipeline_result")

143        # Default file-based storage (backward compatible)
144        if output_path is None:

146

148        path.parent.mkdir(parents=True, exist_ok=True)
149
150        if isinstance(results, pd.DataFrame):
151            results.to_parquet(path)
152        else:
153            import json
154
155            path = path.with_suffix(".json")
156            path.write_text(json.dumps(results, default=str))
157
158        return str(path)
