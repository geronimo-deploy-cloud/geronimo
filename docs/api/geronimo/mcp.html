 12class MCPServer(ABC):
 13    """Base class for MCP servers exposing Geronimo models.
 14    
 15    Subclass this to create an MCP server for your model:
 16    
 17        class IrisMCPServer(MCPServer):
 18            name = "iris-classifier"
 19            version = "1.0.0"
 20            
 21            def get_tools(self) -> list[Tool]:
 22                return [
 23                    Tool(
 24                        name="predict",
 25                        description="Predict iris species",
 26                        input_schema={...},
 27                        handler=self.predict,
 28                    )
 29                ]
 30            
 31            def predict(self, features: dict) -> ToolResult:
 32                result = self.model.predict(features)
 33                return ToolResult.json(result)
 34    """
 35    
 36    name: str = "geronimo-model"
 37    version: str = "1.0.0"
 38    description: str = "Geronimo ML Model MCP Server"
 39    
 40    def __init__(self):
 41        self._tools: dict[str, Tool] = {}
 42        self._initialize_tools()
 43    
 44    def _initialize_tools(self):
 45        """Register tools from get_tools()."""
 46        for tool in self.get_tools():
 47            self._tools[tool.name] = tool
 48    
 49    @abstractmethod
 50    def get_tools(self) -> list[Tool]:
 51        """Return list of tools this server exposes.
 52        
 53        Override this to define your model's tools.
 54        """
 55        pass
 56    
 57    def handle_request(self, request: dict) -> dict:
 58        """Handle an incoming MCP request."""
 59        method = request.get("method", "")
 60        request_id = request.get("id")
 61        params = request.get("params", {})
 62        
 63        if method == "initialize":
 64            return self._handle_initialize(request_id, params)
 65        elif method == "tools/list":
 66            return self._handle_tools_list(request_id)
 67        elif method == "tools/call":
 68            return self._handle_tools_call(request_id, params)
 69        else:
 70            return self._error_response(request_id, -32601, f"Unknown method: {method}")
 71    
 72    def _handle_initialize(self, request_id: int, params: dict) -> dict:
 73        """Handle initialize request."""
 74        return {
 75            "jsonrpc": "2.0",
 76            "id": request_id,
 77            "result": {
 78                "protocolVersion": "2024-11-05",
 79                "capabilities": {
 80                    "tools": {},
 81                },
 82                "serverInfo": {
 83                    "name": self.name,
 84                    "version": self.version,
 85                },
 86            },
 87        }
 88    
 89    def _handle_tools_list(self, request_id: int) -> dict:
 90        """Handle tools/list request."""
 91        tools = [tool.to_mcp_schema() for tool in self._tools.values()]
 92        return {
 93            "jsonrpc": "2.0",
 94            "id": request_id,
 95            "result": {"tools": tools},
 96        }
 97    
 98    def _handle_tools_call(self, request_id: int, params: dict) -> dict:
 99        """Handle tools/call request."""
100        tool_name = params.get("name", "")
101        arguments = params.get("arguments", {})
102        
103        if tool_name not in self._tools:
104            return self._error_response(request_id, -32602, f"Unknown tool: {tool_name}")
105        
106        tool = self._tools[tool_name]
107        result = tool.invoke(arguments)
108        
109        return {
110            "jsonrpc": "2.0",
111            "id": request_id,
112            "result": {
113                "content": result.content,
114                "isError": result.is_error,
115            },
116        }
117    
118    def _error_response(self, request_id: int, code: int, message: str) -> dict:
119        """Create an error response."""
120        return {
121            "jsonrpc": "2.0",
122            "id": request_id,
123            "error": {"code": code, "message": message},
124        }
125    
126    def run_stdio(self):
127        """Run the server using stdio transport.
128        
129        This is the standard transport for local MCP servers.
130        """
131        print(f"Starting MCP server: {self.name} v{self.version}", file=sys.stderr)
132        
133        for line in sys.stdin:
134            line = line.strip()
135            if not line:
136                continue
137            
138            try:
139                request = json.loads(line)
140                response = self.handle_request(request)
141                print(json.dumps(response), flush=True)
142            except json.JSONDecodeError as e:
143                error = self._error_response(None, -32700, f"Parse error: {e}")
144                print(json.dumps(error), flush=True)
145            except Exception as e:
146                error = self._error_response(None, -32603, f"Internal error: {e}")
147                print(json.dumps(error), flush=True)

56    @classmethod
57    def define(cls, name: str, description: str, input_schema: dict):
58        """Decorator to define a tool."""
59        def decorator(func: Callable[..., ToolResult]) -> "Tool":
60            return cls(
61                name=name,
62                description=description,
63                input_schema=input_schema,
64                handler=func,
65            )
66        return decorator
67    
68    def to_mcp_schema(self) -> dict:
69        """Convert to MCP tool schema format."""
70        return {
71            "name": self.name,
72            "description": self.description,
73            "inputSchema": self.input_schema,
74        }

76    def invoke(self, arguments: dict) -> ToolResult:
77        """Invoke the tool with given arguments."""
78        try:

80        except Exception as e:
81            return ToolResult.error(str(e))
